    20→def api(method, path, body=None):
    21→    req = urllib.request.Request(f"https://api.vercel.com{path}", method=method)
    22→    req.add_header("Authorization", f"Bearer {TOKEN}")
    23→    req.add_header("Content-Type", "application/json")
    24→    data = json.dumps(body).encode() if body is not None else None
    25→    try:
    26→        with urllib.request.urlopen(req, data=data, timeout=60) as r:
    27→            return r.status, r.read().decode()
    28→    except urllib.error.HTTPError as e:
    29→        return e.code, e.read().decode()[:600]
